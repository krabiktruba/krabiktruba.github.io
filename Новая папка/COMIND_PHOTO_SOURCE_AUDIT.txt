import sqlite3
from pathlib import Path
from contextlib import contextmanager

DATABASE_PATH = Path(__file__).resolve().parents[2] / "comind_soon.db"


def get_connection() -> sqlite3.Connection:
    connection = sqlite3.connect(DATABASE_PATH)
    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA foreign_keys = ON")
    return connection


@contextmanager
def get_db():
    connection = get_connection()
    try:
        yield connection
        connection.commit()
    except Exception:
        connection.rollback()
        raise
    finally:
        connection.close()

def init_forum():
    with get_db() as db:
        db.executescript("""
            CREATE TABLE IF NOT EXISTS forum_categories (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                slug TEXT NOT NULL UNIQUE,
                name TEXT NOT NULL,
                description TEXT NOT NULL DEFAULT '',
                sort_order INTEGER NOT NULL DEFAULT 0,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
            );

            CREATE TABLE IF NOT EXISTS forum_topics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                category_id INTEGER NOT NULL,
                author_id INTEGER NOT NULL,
                title TEXT NOT NULL,
                content TEXT NOT NULL,
                views INTEGER NOT NULL DEFAULT 0,
                is_pinned INTEGER NOT NULL DEFAULT 0,
                is_locked INTEGER NOT NULL DEFAULT 0,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY(category_id)
                    REFERENCES forum_categories(id),
                FOREIGN KEY(author_id)
                    REFERENCES users(id)
            );

            CREATE TABLE IF NOT EXISTS forum_posts (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                topic_id INTEGER NOT NULL,
                author_id INTEGER NOT NULL,
                content TEXT NOT NULL,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                edited_at TEXT,
                FOREIGN KEY(topic_id)
                    REFERENCES forum_topics(id) ON DELETE CASCADE,
                FOREIGN KEY(author_id)
                    REFERENCES users(id)
            );

            CREATE INDEX IF NOT EXISTS idx_forum_topics_category
            ON forum_topics(category_id, created_at);

            CREATE INDEX IF NOT EXISTS idx_forum_topics_author
            ON forum_topics(author_id);

            CREATE INDEX IF NOT EXISTS idx_forum_posts_topic
            ON forum_posts(topic_id, id);

            CREATE INDEX IF NOT EXISTS idx_forum_posts_author
            ON forum_posts(author_id);
        """)

        for order, (slug, name, description) in enumerate(DEFAULT_CATEGORIES):
            db.execute("""
                INSERT OR IGNORE INTO forum_categories
                (slug, name, description, sort_order)
                VALUES (?, ?, ?, ?)
            """, (slug, name, description, order))

def get_topic(db, topic_id):
    topic = db.execute("""
        SELECT
            t.*,
            u.username AS author,
            COALESCE(u.display_name, '') AS display_name,
            c.name AS category,
            c.slug AS category_slug,
            (SELECT COUNT(*) FROM forum_posts p
             WHERE p.topic_id=t.id) AS replies
        FROM forum_topics t
        JOIN users u ON u.id=t.author_id
        JOIN forum_categories c ON c.id=t.category_id
        WHERE t.id=?
    """, (topic_id,)).fetchone()

    return dict(topic) if topic else None


@router.get("/categories")
def categories(user: dict = Depends(current_user)):
    with get_db() as db:
        rows = db.execute("""
            SELECT
                c.*,
                COUNT(t.id) AS topic_count
            FROM forum_categories c
            LEFT JOIN forum_topics t ON t.category_id=c.id
            GROUP BY c.id
            ORDER BY c.sort_order, c.id
        """).fetchall()

    return [dict(row) for row in rows]


@router.get("/topics")
def topics(
    category_id: int | None = None,
    q: str = Query(default="", max_length=150),
    limit: int = Query(default=30, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
    user: dict = Depends(current_user),
):
    conditions = []
    params = []

    if category_id is not None:
        conditions.append("t.category_id=?")
        params.append(category_id)

    if q.strip():
        conditions.append("(t.title LIKE ? OR t.content LIKE ?)")
        term = "%" + q.strip() + "%"
        params.extend([term, term])

    where = "WHERE " + " AND ".join(conditions) if conditions else ""

    with get_db() as db:
        rows = db.execute(f"""
            SELECT
                t.id,
                t.category_id,
                t.author_id,
                t.title,
                t.content,
                t.views,
                t.is_pinned,
                t.is_locked,
                t.created_at,
                t.updated_at,
                u.username AS author,
                c.name AS category,
                c.slug AS category_slug,
                (SELECT COUNT(*) FROM forum_posts p
                 WHERE p.topic_id=t.id) AS replies
            FROM forum_topics t
            JOIN users u ON u.id=t.author_id
            JOIN forum_categories c ON c.id=t.category_id
            {where}
            ORDER BY t.is_pinned DESC, t.created_at DESC, t.id DESC
            LIMIT ? OFFSET ?
        """, (*params, limit, offset)).fetchall()

        total = db.execute(f"""
            SELECT COUNT(*) AS total
            FROM forum_topics t
            {where}
        """, params).fetchone()["total"]

    return {
        "items": [dict(row) for row in rows],
        "total": total,
        "limit": limit,
        "offset": offset,
    }


@router.post("/topics", status_code=201)
def create_topic(
    data: TopicCreate,
    user: dict = Depends(current_user),
):
    with get_db() as db:
        category = db.execute(
            "SELECT id FROM forum_categories WHERE id=?",
            (data.category_id,)
        ).fetchone()

        if category is None:
            raise HTTPException(404, "Категория не найдена")

        cursor = db.execute("""
            INSERT INTO forum_topics
            (category_id, author_id, title, content)
            VALUES (?, ?, ?, ?)
        """, (
            data.category_id,
            user["id"],
            data.title.strip(),
            data.content.strip(),
        ))

        topic = get_topic(db, cursor.lastrowid)

    return topic

        result = get_topic(db, topic_id)

    return result

async function request<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers);
  headers.set("Accept", "application/json");

  if (options.body !== undefined && options.body !== null) {
    headers.set("Content-Type", "application/json");
  }

  const token = localStorage.getItem("comind_access_token");
  if (token) headers.set("Authorization", `Bearer ${token}`);

  let response: Response;

  try {
    response = await fetch(`${API_URL}${path}`, {
      ...options,
      headers,
    });
  } catch {
    throw new Error(
      `Не удалось подключиться к COMIND API (${API_URL}). Проверь backend.`,
    );
  }

  if (response.status === 204) return undefined as T;

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    const detail = (data as { detail?: unknown })?.detail;
    const message = Array.isArray(detail)
      ? detail.map((item: { msg?: string }) => item.msg || "").filter(Boolean).join("; ")
      : typeof detail === "string"
        ? detail
        : `Ошибка HTTP ${response.status}`;

    if (response.status === 401) {
      throw new Error("Сессия истекла. Войди в аккаунт заново.");
    }

    throw new Error(message);
  }

  return data as T;
}

function list<T>(value: unknown, label: string): T[] {
  if (Array.isArray(value)) return value as T[];

  if (value && typeof value === "object") {
    const obj = value as { items?: unknown; results?: unknown };
    if (Array.isArray(obj.items)) return obj.items as T[];
    if (Array.isArray(obj.results)) return obj.results as T[];
  }

  throw new Error(`Неожиданный формат ответа API: ${label}`);
}

function body(data: unknown): string {
  return JSON.stringify(data);
