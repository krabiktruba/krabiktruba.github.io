):
    message = repo.get_message(message_id)
    if message is None:
        raise HTTPException(status_code=404, detail="Сообщение не найдено")

    if not repo.is_member(message["community_id"], user["id"]):
        raise HTTPException(404, detail="Message not found")

    is_author = int(message["user_id"]) == int(user["id"])
    is_owner = repo.is_owner(message["community_id"], user["id"])
    is_admin = bool(user.get("is_admin", 0))

    if not is_author and not is_owner and not is_admin:
        raise HTTPException(
            status_code=403,
            detail="Редактировать сообщение может его автор, владелец сообщества или администратор",
        )

    content = data.content.strip()
    if not content:
        raise HTTPException(status_code=422, detail="Сообщение не может быть пустым")

    updated = repo.update_message(message_id, content)
    if updated is None:
        raise HTTPException(status_code=404, detail="Сообщение не найдено")
    return updated

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field

from app.core.database import get_db
from app.api.routes.auth import current_user


router = APIRouter(prefix="/api/dm", tags=["Direct Messages"])


class ConversationCreate(BaseModel):
    friend_id: int


class MessageCreate(BaseModel):
    content: str = Field(min_length=1, max_length=10000)


def init_dm():
    with get_db() as db:
        db.execute("""
            CREATE TABLE IF NOT EXISTS dm_conversations (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_low_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
                user_high_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                UNIQUE(user_low_id, user_high_id),
                CHECK(user_low_id < user_high_id)
            )
        """)

        db.execute("""
            CREATE TABLE IF NOT EXISTS dm_participants (
                conversation_id INTEGER NOT NULL
                    REFERENCES dm_conversations(id) ON DELETE CASCADE,
                user_id INTEGER NOT NULL
                    REFERENCES users(id) ON DELETE CASCADE,
                last_read_message_id INTEGER NOT NULL DEFAULT 0,
                joined_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
                PRIMARY KEY(conversation_id, user_id)
            )
        """)

        db.execute("""
            CREATE TABLE IF NOT EXISTS dm_messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                conversation_id INTEGER NOT NULL
                    REFERENCES dm_conversations(id) ON DELETE CASCADE,
                sender_id INTEGER NOT NULL
                    REFERENCES users(id) ON DELETE CASCADE,
                content TEXT NOT NULL,
                created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
            )
        """)

        participant_columns = {
            row["name"]
            for row in db.execute("PRAGMA table_info(dm_participants)")
        }

        if "cleared_before_message_id" not in participant_columns:
            db.execute("""
                ALTER TABLE dm_participants
                ADD COLUMN cleared_before_message_id INTEGER NOT NULL DEFAULT 0
            """)

        if "hidden_at" not in participant_columns:
            db.execute("""
                ALTER TABLE dm_participants
                ADD COLUMN hidden_at TEXT
            """)

        message_columns = {
            row["name"]
            for row in db.execute("PRAGMA table_info(dm_messages)")
        }

        if "edited_at" not in message_columns:
            db.execute("ALTER TABLE dm_messages ADD COLUMN edited_at TEXT")

        if "deleted_at" not in message_columns:
            db.execute("ALTER TABLE dm_messages ADD COLUMN deleted_at TEXT")

        db.execute("""
            CREATE INDEX IF NOT EXISTS idx_dm_messages_conversation
            ON dm_messages(conversation_id, id)
        """)


def _participant(db, conversation_id, user_id):
    return db.execute("""
        SELECT *
        FROM dm_participants
        WHERE conversation_id=? AND user_id=?
    """, (conversation_id, user_id)).fetchone()


def _get_conversation(db, conversation_id, user_id):
    row = db.execute("""
        SELECT *
        FROM dm_conversations
        WHERE id=?
    """, (conversation_id,)).fetchone()

    if row is None:
        raise HTTPException(404, "Диалог не найден")

    if not _participant(db, conversation_id, user_id):
        raise HTTPException(403, "Нет доступа к диалогу")

    friend_id = (
        row["user_high_id"]
        if row["user_low_id"] == user_id
        else row["user_low_id"]
    )

    return row, friend_id


def _blocked(db, a, b):
    return db.execute("""
        SELECT 1
        FROM user_blocks
        WHERE (user_id=? AND blocked_user_id=?)
           OR (user_id=? AND blocked_user_id=?)
        LIMIT 1
    """, (a, b, b, a)).fetchone() is not None


def _blocked_by_me(db, user_id, friend_id):
    return db.execute("""
        SELECT 1 FROM user_blocks
        WHERE user_id=? AND blocked_user_id=?
    """, (user_id, friend_id)).fetchone() is not None


def _blocked_by_them(db, user_id, friend_id):
    return _blocked_by_me(db, friend_id, user_id)


def _friends(db, a, b):
    return db.execute("""
        SELECT 1
        FROM friend_requests
        WHERE status='accepted'
          AND (
            (sender_id=? AND receiver_id=?)
            OR
            (sender_id=? AND receiver_id=?)
          )
        LIMIT 1
    """, (a, b, b, a)).fetchone() is not None


def _can_send(db, a, b):
    return _friends(db, a, b) and not _blocked(db, a, b)


def _conversation(db, a, b):
    low, high = sorted((a, b))

    db.execute("""
        INSERT OR IGNORE INTO dm_conversations(user_low_id, user_high_id)
        VALUES (?, ?)
    """, (low, high))

    row = db.execute("""
        SELECT id FROM dm_conversations
        WHERE user_low_id=? AND user_high_id=?
    """, (low, high)).fetchone()

    conversation_id = row["id"]

    db.execute("""
        INSERT OR IGNORE INTO dm_participants(conversation_id, user_id)
        VALUES (?, ?), (?, ?)
    """, (conversation_id, a, conversation_id, b))

    return conversation_id


def _message_dict(row):
    return {
        "id": row["id"],
        "conversation_id": row["conversation_id"],
        "sender_id": row["sender_id"],
        "sender_username": row["sender_username"],
        "content": row["content"],
        "created_at": row["created_at"],
        "edited_at": row["edited_at"],
        "is_deleted": bool(row["deleted_at"]),
    }


def _fetch_message(db, message_id, conversation_id):
    return db.execute("""
        SELECT
            m.id,
            m.conversation_id,
            m.sender_id,
            u.username AS sender_username,
            m.content,
            m.created_at,
            m.edited_at,
            m.deleted_at
        FROM dm_messages m
        JOIN users u ON u.id=m.sender_id
        WHERE m.id=? AND m.conversation_id=?
    """, (message_id, conversation_id)).fetchone()


@router.get("/conversations")
def list_conversations(user=Depends(current_user)):
    uid = user["id"]

    with get_db() as db:
        rows = db.execute("""
            SELECT
                c.id,
                CASE
                    WHEN c.user_low_id=? THEN c.user_high_id
                    ELSE c.user_low_id
                END AS friend_id,
                u.username AS friend_username,
                p.last_read_message_id,
                p.cleared_before_message_id,
                (
                    SELECT CASE
                        WHEN m.deleted_at IS NOT NULL
                        THEN 'Сообщение удалено'
                        ELSE m.content
                    END
                    FROM dm_messages m
                    WHERE m.conversation_id=c.id
                      AND m.id > p.cleared_before_message_id
                    ORDER BY m.id DESC
                    LIMIT 1
                ) AS last_message,
                (
                    SELECT m.id
                    FROM dm_messages m
                    WHERE m.conversation_id=c.id
                      AND m.id > p.cleared_before_message_id
                    ORDER BY m.id DESC
                    LIMIT 1
                ) AS last_message_id,
                (
                    SELECT m.created_at
                    FROM dm_messages m
                    WHERE m.conversation_id=c.id
                      AND m.id > p.cleared_before_message_id
                    ORDER BY m.id DESC
                    LIMIT 1
                ) AS last_message_at
            FROM dm_conversations c
            JOIN dm_participants p
              ON p.conversation_id=c.id AND p.user_id=?
            JOIN users u
              ON u.id=CASE
                WHEN c.user_low_id=? THEN c.user_high_id
                ELSE c.user_low_id
              END
            WHERE p.hidden_at IS NULL
            ORDER BY COALESCE(last_message_id, 0) DESC, c.id DESC
        """, (uid, uid, uid)).fetchall()

        result = []

        for row in rows:
            item = dict(row)
            friend_id = item["friend_id"]

            item["is_blocked_by_me"] = _blocked_by_me(
                db, uid, friend_id
            )
            item["is_blocked_by_them"] = _blocked_by_them(
                db, uid, friend_id
            )
            item["can_send"] = _can_send(db, uid, friend_id)

            result.append(item)

        return result


@router.post("/conversations")
def create_conversation(
    data: ConversationCreate,
    user=Depends(current_user)
):
    uid = user["id"]
    friend_id = data.friend_id

    if friend_id == uid:
        raise HTTPException(400, "Нельзя создать диалог с самим собой")

    with get_db() as db:
        if not db.execute(
            "SELECT id FROM users WHERE id=?",
            (friend_id,)
        ).fetchone():
            raise HTTPException(404, "Пользователь не найден")

        if _blocked(db, uid, friend_id):
            raise HTTPException(403, "Пользователь заблокирован")

        if not _friends(db, uid, friend_id):
            raise HTTPException(
                403,
                "Личные сообщения доступны только друзьям"
            )

        conversation_id = _conversation(db, uid, friend_id)

        db.execute("""
            UPDATE dm_participants
            SET hidden_at=NULL
            WHERE conversation_id=? AND user_id=?
        """, (conversation_id, uid))

        return {"id": conversation_id, "friend_id": friend_id}

def member(db, cid, uid):
    return db.execute(
        "SELECT role FROM community_members WHERE community_id=? AND user_id=?",
        (cid, uid)
    ).fetchone()

def require_member(db, cid, uid):
    m = member(db, cid, uid)
    if not m:
        raise HTTPException(403, "Вы не участник сообщества")
    return m["role"]

def require_manager(db, cid, uid):
    role = require_member(db, cid, uid)
    if role not in ("owner", "admin", "moderator"):
        raise HTTPException(403, "Недостаточно прав")
    return role

def search(q: str=Query(min_length=2,max_length=100), user=Depends(current_user)):

    with get_db() as db:

          SELECT c.id,c.name,c.description,c.owner_id,
          (SELECT COUNT(*) FROM community_members m WHERE m.community_id=c.id) member_count
          FROM communities c
          WHERE c.name LIKE ? ESCAPE '\\'
            AND (
              COALESCE(c.visibility, 'public')='public'
              OR EXISTS (
                SELECT 1 FROM community_members cm
                WHERE cm.community_id=c.id AND cm.user_id=?
              )
            )
          ORDER BY c.id DESC LIMIT 20
        """,(term,user["id"]))]
        users=[dict(r) for r in db.execute("""
          SELECT id,username,display_name,avatar_url FROM users
          WHERE username LIKE ? ESCAPE '\\' OR display_name LIKE ? ESCAPE '\\' LIMIT 20
        """,(term,term))]
        messages=[dict(r) for r in db.execute("""
          SELECT m.id,m.channel_id,m.user_id,u.username,m.content,m.created_at,c.community_id
          FROM messages m JOIN users u ON u.id=m.user_id JOIN channels c ON c.id=m.channel_id
          JOIN community_members cm ON cm.community_id=c.community_id AND cm.user_id=?
          WHERE m.content LIKE ? ESCAPE '\\'
            AND (
              COALESCE(c.visibility, 'public')='public'
              OR EXISTS (
                SELECT 1 FROM channel_members chm
                WHERE chm.channel_id=c.id AND chm.user_id=?
              )
              OR EXISTS (
                SELECT 1 FROM community_members own
                WHERE own.community_id=c.community_id
                  AND own.user_id=? AND own.role='owner'
              )
            )
            ORDER BY m.id DESC LIMIT 30
        """,(user["id"],term,user["id"],user["id"]))]
        return {"communities":communities,"users":users,"messages":messages}

def require_voice_channel_access(db, channel, user_id: int, is_admin: bool = False):
    if channel is None or channel["channel_type"] != "voice":
        raise HTTPException(404, "Voice room not found")
    if is_admin:
        return

    membership = db.execute(
        "SELECT role FROM community_members WHERE community_id=? AND user_id=?",
        (channel["community_id"], user_id),
    ).fetchone()

    if membership is None:
        raise HTTPException(403, "Community membership required")

    if channel["visibility"] == "private" and membership["role"] != "owner":
        access = db.execute(
            "SELECT 1 FROM channel_members WHERE channel_id=? AND user_id=?",
            (channel["id"], user_id),
        ).fetchone()
        if access is None:
            raise HTTPException(403, "Private voice room access denied")

@router.get("/api/friends")
def get_friends(user=Depends(current_user)):
    with get_db() as db:
        uid = user["id"]

        friends = [dict(row) for row in db.execute("""
            SELECT u.id, u.username, u.display_name, u.avatar_url, u.status
            FROM friend_requests fr
            JOIN users u ON u.id = CASE
                WHEN fr.sender_id = ? THEN fr.receiver_id
                ELSE fr.sender_id
            END
            WHERE (fr.sender_id = ? OR fr.receiver_id = ?)
              AND fr.status = 'accepted'
            ORDER BY u.username COLLATE NOCASE
        """, (uid, uid, uid))]

        incoming = [dict(row) for row in db.execute("""
            SELECT fr.id AS request_id, u.id, u.username,
                   u.display_name, u.avatar_url, u.status
            FROM friend_requests fr
            JOIN users u ON u.id = fr.sender_id
            WHERE fr.receiver_id = ? AND fr.status = 'pending'
            ORDER BY fr.id DESC
        """, (uid,))]

        outgoing = [dict(row) for row in db.execute("""
            SELECT fr.id AS request_id, u.id, u.username,
                   u.display_name, u.avatar_url, u.status
            FROM friend_requests fr
            JOIN users u ON u.id = fr.receiver_id
            WHERE fr.sender_id = ? AND fr.status = 'pending'
            ORDER BY fr.id DESC
        """, (uid,))]

        return {"friends": friends, "incoming": incoming, "outgoing": outgoing}


@router.post("/api/friends/requests", status_code=201)
def create_friend_request(data: FriendRequestCreate, user=Depends(current_user)):
    with get_db() as db:
        target = db.execute(
            "SELECT id, username FROM users WHERE username = ? COLLATE NOCASE",
            (data.username.strip(),),
        ).fetchone()

        if target is None:
            raise HTTPException(404, "Пользователь не найден")
        if target["id"] == user["id"]:
            raise HTTPException(400, "Нельзя добавить самого себя")

        sender_id = user["id"]
        receiver_id = target["id"]

        accepted = db.execute("""
            SELECT id FROM friend_requests
            WHERE status = 'accepted'
              AND ((sender_id = ? AND receiver_id = ?)
                OR (sender_id = ? AND receiver_id = ?))
            LIMIT 1
        """, (sender_id, receiver_id, receiver_id, sender_id)).fetchone()

        if accepted:
            return {"status": "already_friends", "username": target["username"]}

        reverse = db.execute("""
            SELECT id, status FROM friend_requests
            WHERE sender_id = ? AND receiver_id = ?
        """, (receiver_id, sender_id)).fetchone()

        if reverse and reverse["status"] == "pending":
            db.execute("""
                UPDATE friend_requests
                SET status = 'accepted', updated_at = CURRENT_TIMESTAMP
                WHERE id = ?
            """, (reverse["id"],))
            return {"status": "accepted", "username": target["username"]}

        existing = db.execute("""
            SELECT id, status FROM friend_requests
            WHERE sender_id = ? AND receiver_id = ?
        """, (sender_id, receiver_id)).fetchone()

        if existing and existing["status"] == "pending":
            return {"status": "already_pending", "username": target["username"]}

        if existing:
            db.execute("""
                UPDATE friend_requests
                SET status = 'pending', updated_at = CURRENT_TIMESTAMP
                WHERE id = ?
            """, (existing["id"],))
        else:
            db.execute("""
                INSERT INTO friend_requests(sender_id, receiver_id, status)
                VALUES (?, ?, 'pending')
            """, (sender_id, receiver_id))

        return {"status": "pending", "username": target["username"]}


@router.patch("/api/friends/requests/{request_id}")
def respond_friend_request(
    request_id: int,
    data: FriendRequestAction,
    user=Depends(current_user),
):
    with get_db() as db:
        request = db.execute("""
            SELECT id, sender_id, receiver_id, status
            FROM friend_requests WHERE id = ?
        """, (request_id,)).fetchone()

        if request is None:
            raise HTTPException(404, "Запрос дружбы не найден")
        if request["receiver_id"] != user["id"]:
            raise HTTPException(403, "Можно отвечать только на входящие запросы")
        if request["status"] != "pending":
            raise HTTPException(409, "Запрос уже обработан")

        status = "accepted" if data.action == "accept" else "rejected"
        db.execute("""
            UPDATE friend_requests
            SET status = ?, updated_at = CURRENT_TIMESTAMP
            WHERE id = ?
        """, (status, request_id))

        return {"status": status, "request_id": request_id}

import secrets
import time
from typing import Any

from fastapi import APIRouter, Depends, HTTPException, WebSocket, WebSocketDisconnect

from app.api.routes.auth import current_user
from app.repositories import channel_repository as channel_repo

router = APIRouter(prefix="/api/voice", tags=["Voice / WebRTC"])

TICKETS: dict[str, dict[str, Any]] = {}
ROOMS: dict[int, dict[str, dict[str, Any]]] = {}
TICKET_TTL = 60


@router.post("/{channel_id}/ticket")
def create_voice_ticket(channel_id: int, user: dict = Depends(current_user)):
    channel = channel_repo.get_channel(channel_id)
    if channel is None:
        raise HTTPException(404, "Voice channel not found")
    if channel["channel_type"] != "voice":
        raise HTTPException(400, "This is not a voice channel")
    if not channel_repo.can_access_channel(
        channel_id, user["id"], bool(user.get("is_admin", False))
    ):
        raise HTTPException(403, "No access to this voice channel")

    ticket = secrets.token_urlsafe(32)
    TICKETS[ticket] = {
        "channel_id": channel_id,
        "user_id": user["id"],
        "username": user["username"],
        "is_admin": bool(user.get("is_admin", False)),
        "expires_at": time.time() + TICKET_TTL,
    }
    return {"ticket": ticket, "expires_in": TICKET_TTL}


@router.websocket("/ws")
async def voice_signalling(websocket: WebSocket):
    ticket = websocket.query_params.get("ticket", "")
    grant = TICKETS.pop(ticket, None)

    if not grant or grant["expires_at"] < time.time():
        await websocket.close(code=1008, reason="Invalid or expired ticket")
        return

    channel_id = int(grant["channel_id"])
    channel = channel_repo.get_channel(channel_id)

    if (
        channel is None
        or channel["channel_type"] != "voice"
        or not channel_repo.can_access_channel(
            channel_id,
            int(grant["user_id"]),
            bool(grant.get("is_admin", False)),
        )
    ):
        await websocket.close(code=1008, reason="No access to voice channel")
        return

    await websocket.accept()

    peer_id = secrets.token_hex(8)
    username = str(grant["username"])
    room = ROOMS.setdefault(channel_id, {})

    existing = [
        {"peer_id": other_id, "username": peer["username"]}
        for other_id, peer in room.items()
    ]

    room[peer_id] = {
        "socket": websocket,
        "user_id": int(grant["user_id"]),
        "username": username,
    }

    await websocket.send_json({
        "type": "hello",
        "peer_id": peer_id,
        "peers": existing,
    })

    for other_id, peer in list(room.items()):
        if other_id == peer_id:
            continue
        try:
            await peer["socket"].send_json({
                "type": "peer-joined",
                "peer_id": peer_id,
                "username": username,
            })
        except Exception:
            pass

    try:
        while True:
            message = await websocket.receive_json()

            if not isinstance(message, dict):
                continue

            kind = message.get("type")
            if kind == "leave":
                break

            if kind not in {"offer", "answer", "ice-candidate"}:
                continue

            target_id = message.get("to")
            if not isinstance(target_id, str) or target_id == peer_id:
                continue

            target = room.get(target_id)
            if target is None:
                continue

            outgoing: dict[str, Any] = {
                "type": kind,
                "from": peer_id,
                "username": username,
            }

            if kind in {"offer", "answer"}:
                sdp = message.get("sdp")
                if not isinstance(sdp, dict):
                    continue
                if sdp.get("type") != kind:
                    continue
                if not isinstance(sdp.get("sdp"), str) or len(sdp["sdp"]) > 150000:
                    continue
                outgoing["sdp"] = {"type": kind, "sdp": sdp["sdp"]}
            else:
                candidate = message.get("candidate")
                if not isinstance(candidate, dict):
                    continue
                if not isinstance(candidate.get("candidate"), str):
                    continue
                outgoing["candidate"] = candidate

            try:
                await target["socket"].send_json(outgoing)
            except Exception:
                pass

    except (WebSocketDisconnect, RuntimeError):
        pass
    finally:
        current_room = ROOMS.get(channel_id)
        if current_room is not None:
            current_room.pop(peer_id, None)

            for peer in list(current_room.values()):
                try:
                    await peer["socket"].send_json({
                        "type": "peer-left",
                        "peer_id": peer_id,
                    })
                except Exception:
                    pass

            if not current_room:
                ROOMS.pop(channel_id, None)

def list_communities(user_id: int | None = None) -> list[dict]:
    with get_db() as connection:
        uid = int(user_id) if user_id is not None else -1

        rows = connection.execute(
            """
            SELECT
                c.*,
                u.username AS owner_username,

                (
                    SELECT COUNT(*)
                    FROM community_members m
                    WHERE m.community_id=c.id
                ) AS member_count,

                CASE WHEN EXISTS (
                    SELECT 1
                    FROM community_members m
                    WHERE m.community_id=c.id AND m.user_id=?
                ) THEN 1 ELSE 0 END AS is_member

            FROM communities c
            JOIN users u ON u.id=c.owner_id

            WHERE
                COALESCE(c.visibility, 'public')='public'
                OR EXISTS (
                    SELECT 1
                    FROM community_members m
                    WHERE m.community_id=c.id AND m.user_id=?
                )

            ORDER BY c.id DESC
            """,
            (uid, uid),
        ).fetchall()

        return [dict(row) for row in rows]

  headers.set("Accept", "application/json");

  if (options.body !== undefined && options.body !== null) {
    headers.set("Content-Type", "application/json");
  }

  const token = localStorage.getItem("comind_access_token");
  if (token) headers.set("Authorization", `Bearer ${token}`);

  let response: Response;

  try {
    response = await fetch(`${API_URL}${path}`, {
      ...options,
      headers,
    });
  } catch {
    throw new Error(
      `Не удалось подключиться к COMIND API (${API_URL}). Проверь backend.`,
    );
  }

  if (response.status === 204) return undefined as T;

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    const detail = (data as { detail?: unknown })?.detail;
    const message = Array.isArray(detail)
      ? detail.map((item: { msg?: string }) => item.msg || "").filter(Boolean).join("; ")
      : typeof detail === "string"
        ? detail
        : `Ошибка HTTP ${response.status}`;

    if (response.status === 401) {
      throw new Error("Сессия истекла. Войди в аккаунт заново.");
    }

    throw new Error(message);
  }

  return data as T;
}

function list<T>(value: unknown, label: string): T[] {
  if (Array.isArray(value)) return value as T[];

  if (value && typeof value === "object") {
    const obj = value as { items?: unknown; results?: unknown };
    if (Array.isArray(obj.items)) return obj.items as T[];
    if (Array.isArray(obj.results)) return obj.results as T[];
  }

  throw new Error(`Неожиданный формат ответа API: ${label}`);
}

function body(data: unknown): string {
  return JSON.stringify(data);
}

import { useEffect, useState } from "react";
import type { FormEvent } from "react";
import "./App.css";
import MainWindow from "./MainWindow";

const API_URL = (
  import.meta.env.VITE_API_URL || "http://127.0.0.1:8000"
).replace(/\/$/, "");

type User = {
  id: number;
  username: string;
  email: string;
  created_at: string;
  is_admin?: boolean | number;
};

type Mode = "login" | "register";

async function apiRequest<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  const response = await fetch(`${API_URL}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(options.headers || {}),
    },
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    const detail = data.detail;
    const message = Array.isArray(detail)
      ? detail.map((item: { msg?: string }) => item.msg || "").join(", ")
      : typeof detail === "string"
        ? detail
        : `Ошибка сервера: ${response.status}`;

    throw new Error(message || "Не удалось выполнить запрос");
  }

  return data as T;
}

export default function App() {
  const [mode, setMode] = useState<Mode>("login");
  const [username, setUsername] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(false);
  const [checkingSession, setCheckingSession] = useState(true);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");

  useEffect(() => {
    let cancelled = false;

    async function restoreSession() {
      const token = localStorage.getItem("comind_access_token");

      if (!token) {
        if (!cancelled) setCheckingSession(false);
        return;
      }

      try {
        const profile = await apiRequest<User>("/api/auth/me", {
          headers: { Authorization: `Bearer ${token}` },
        });

        if (!cancelled) setUser(profile);
      } catch {
        localStorage.removeItem("comind_access_token");
      } finally {
        if (!cancelled) setCheckingSession(false);
      }
    }

    void restoreSession();
    return () => {
      cancelled = true;
    };
  }, []);

  function changeMode(nextMode: Mode) {
    setMode(nextMode);
    setError("");
    setNotice("");
  }

  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setError("");
    setNotice("");

    const normalizedEmail = email.trim().toLowerCase();

    if (mode === "register") {
      if (!/^[a-zA-Z0-9_]{3,32}$/.test(username)) {
        setError("Имя пользователя: 3–32 символа, только латинские буквы, цифры и _.");
        return;
      }

      if (password.length < 8 || password.length > 128) {
        setError("Пароль должен содержать от 8 до 128 символов.");
        return;
      }
    }

    setLoading(true);

    try {
      if (mode === "register") {
        await apiRequest<User>("/api/auth/register", {
          method: "POST",
          body: JSON.stringify({
            username: username.trim(),
            email: normalizedEmail,
            password,
          }),
        });
      }

      const tokenResponse = await apiRequest<{
        access_token: string;
        token_type: string;
      }>("/api/auth/login", {
        method: "POST",
        body: JSON.stringify({
          email: normalizedEmail,
          password,
        }),
      });

      localStorage.setItem("comind_access_token", tokenResponse.access_token);

      const profile = await apiRequest<User>("/api/auth/me", {
        headers: {
          Authorization: `Bearer ${tokenResponse.access_token}`,
        },
      });

      setUser(profile);
      setPassword("");
      setNotice("");
    } catch (err) {
      setError(
        err instanceof TypeError
          ? "Не удаётся подключиться к серверу. Проверь, запущен ли backend."
          : err instanceof Error
            ? err.message
            : "Произошла ошибка.",
      );
    } finally {
      setLoading(false);
    }
  }

  function logout() {
    localStorage.removeItem("comind_access_token");
    setUser(null);
    setPassword("");
    setError("");
    setNotice("Вы вышли из аккаунта.");
    setMode("login");
  }

  if (checkingSession) {
    return (
      <main className="app-shell">
        <div className="loading-card">
          <div className="brand-mark">C</div>
          <p>Подключение к COMIND…</p>
        </div>
      </main>
    );
  }

  if (user) {
    return <MainWindow username={user.username} userId={user.id} userIsAdmin={Boolean(user.is_admin)} onLogout={logout} />;
  }

  return (
    <main className="app-shell">
      <div className="ambient ambient-one" />
      <div className="ambient ambient-two" />
      <section className="auth-card">
        <div className="brand">
          <div className="brand-mark">C</div>
          <span>COMIND</span>
        </div>
        <div className="auth-heading">
          <p className="eyebrow">ТВОЁ ИГРОВОЕ СООБЩЕСТВО</p>
          <h1>{mode === "login" ? "С возвращением" : "Создай аккаунт"}</h1>
          <p className="subtitle">
            {mode === "login" ? "Войди, чтобы продолжить." : "Присоединяйся к своему сообществу."}
          </p>
        </div>
        <div className="mode-switch">
          <button type="button" className={mode === "login" ? "mode-active" : ""} onClick={() => changeMode("login")}>
            Вход
          </button>
          <button type="button" className={mode === "register" ? "mode-active" : ""} onClick={() => changeMode("register")}>
            Регистрация
          </button>
        </div>
        <form className="auth-form" onSubmit={handleSubmit}>
          {mode === "register" && (
            <label className="field">
              <span>Имя пользователя</span>
              <input type="text" placeholder="Например, player_01" value={username} onChange={(event) => setUsername(event.target.value)} autoComplete="username" minLength={3} maxLength={32} required />
            </label>
          )}
          <label className="field">
            <span>Email</span>
            <input type="email" placeholder="you@example.com" value={email} onChange={(event) => setEmail(event.target.value)} autoComplete="email" required />
          </label>
          <label className="field">
            <span>Пароль</span>
            <div className="password-wrapper">
              <input type={showPassword ? "text" : "password"} placeholder={mode === "register" ? "Минимум 8 символов" : "Введи свой пароль"} value={password} onChange={(event) => setPassword(event.target.value)} autoComplete={mode === "login" ? "current-password" : "new-password"} minLength={mode === "register" ? 8 : 1} maxLength={128} required />
              <button className="password-toggle" type="button" onClick={() => setShowPassword((value) => !value)} aria-label={showPassword ? "Скрыть пароль" : "Показать пароль"}>
                {showPassword ? "Скрыть" : "Показать"}
              </button>
            </div>
          </label>
          {mode === "login" && <div className="form-extra"><span>Защищённое подключение</span></div>}
          {error && <div className="message message-error" role="alert">{error}</div>}
          {notice && <div className="message message-success">{notice}</div>}
          <button className="primary-button" type="submit" disabled={loading}>
            {loading ? "Подожди…" : mode === "login" ? "Войти в COMIND" : "Создать аккаунт"}
          </button>

function showDialog(
  config: DialogConfig
): Promise<string | boolean | null> {
  ensureDialogStyles();

  return new Promise((resolve) => {
    if (activeDialog) {
      activeDialog.finish(null);
    }

    const previousFocus =
      document.activeElement instanceof HTMLElement
        ? document.activeElement
        : null;

    const backdrop = document.createElement("div");
    backdrop.className = "comind-dialog-backdrop";

    const panel = document.createElement("section");
    panel.className = "comind-dialog";
    panel.setAttribute("role", "dialog");
    panel.setAttribute("aria-modal", "true");
    panel.setAttribute("aria-labelledby", "comind-dialog-title");

    const form = document.createElement("form");
    form.noValidate = true;

    const heading = document.createElement("h2");
    heading.className = "comind-dialog-title";
    heading.id = "comind-dialog-title";
    heading.textContent = config.title;
    form.appendChild(heading);

    if (config.message) {
      const message = document.createElement("p");
      message.className = "comind-dialog-message";
      message.textContent = config.message;
      form.appendChild(message);
    }

    let field:
      | HTMLInputElement
      | HTMLTextAreaElement
      | HTMLSelectElement
      | null = null;

    if (config.kind === "text") {
      const input = document.createElement("input");
      input.type = "text";
      input.className = "comind-dialog-field";
      input.value = config.defaultValue ?? "";
      input.placeholder = config.placeholder ?? "";
      input.maxLength = 150;
      input.autocomplete = "off";
      field = input;
    }

    if (config.kind === "textarea") {
      const area = document.createElement("textarea");
      area.className = "comind-dialog-field";
      area.value = config.defaultValue ?? "";
      area.placeholder = config.placeholder ?? "";
      area.rows = config.rows ?? 4;
      area.maxLength = 2000;
      field = area;
    }

    if (config.kind === "select") {
      const select = document.createElement("select");
      select.className = "comind-dialog-field";

      for (const item of config.options) {
        const option = document.createElement("option");
        option.value = item.value;
        option.textContent = item.label;
        select.appendChild(option);
      }

      if (config.defaultValue !== undefined) {
        select.value = config.defaultValue;
      }

      field = select;
    }

    if (field) {
      field.setAttribute("aria-label", config.title);
      form.appendChild(field);
    }

    const actions = document.createElement("div");
    actions.className = "comind-dialog-actions";

    const cancel = document.createElement("button");
    cancel.type = "button";
    cancel.className = "comind-dialog-button comind-dialog-cancel";
    cancel.textContent = config.cancelText ?? "??????";

    const submit = document.createElement("button");
    submit.type = "submit";
    submit.className = "comind-dialog-button comind-dialog-submit";
    submit.textContent = config.confirmText ?? "???????????";

    if (config.kind === "confirm" && config.danger) {
      submit.classList.add("comind-dialog-danger");
    }

    actions.append(cancel, submit);
    form.appendChild(actions);
    panel.appendChild(form);
    backdrop.appendChild(panel);

    let settled = false;

    const cancelValue =
      config.kind === "confirm" ? false : null;

    const finish = (value: string | boolean | null) => {
      if (settled) return;
      settled = true;

      document.removeEventListener("keydown", onKeyDown, true);
      backdrop.remove();

      if (activeDialog?.root === backdrop) {
        activeDialog = null;
      }

      previousFocus?.focus();
      resolve(value);
    };

    const onKeyDown = (event: KeyboardEvent) => {
      if (event.key === "Escape") {
        event.preventDefault();
        finish(cancelValue);
      }
    };

    activeDialog = {
      root: backdrop,
      finish,
    };

    cancel.addEventListener("click", () => finish(cancelValue));

    form.addEventListener("submit", (event) => {
      event.preventDefault();

      if (config.kind === "confirm") {
        finish(true);
      } else {
        finish(field?.value ?? "");
      }
    });

    backdrop.addEventListener("mousedown", (event) => {
      if (event.target === backdrop) {
        finish(cancelValue);
      }
    });

    document.addEventListener("keydown", onKeyDown, true);
    document.body.appendChild(backdrop);

    field?.focus();
    if (field instanceof HTMLInputElement) {
      field.select();
    }
  });
}

export function appPrompt(options: {

async function dmRequest<T>(
  path: string,
  options: RequestInit = {}
): Promise<T> {
  const token = localStorage.getItem("comind_access_token");

  const response = await fetch(`${DM_API}/api/dm${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(options.headers || {}),
    },
  });

  if (!response.ok) {
    const body = await response.json().catch(() => ({}));
    throw new Error(body.detail || `Ошибка ${response.status}`);
  }

  if (response.status === 204) return undefined as T;
  return response.json();
}

  const [conversations, setConversations] = useState<Conversation[]>([]);
  const [active, setActive] = useState<Conversation | null>(null);
  const [messages, setMessages] = useState<DirectMessageItem[]>([]);
  const [draft, setDraft] = useState("");
  const [friendIdInput, setFriendIdInput] = useState("");
  const [editingId, setEditingId] = useState<number | null>(null);
  const [editingText, setEditingText] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  const bottomRef = useRef<HTMLDivElement>(null);
  const messagesRef = useRef<Record<number, number>>({});

  const loadConversations = useCallback(async () => {
    try {
      const result = await dmRequest<Conversation[]>("/conversations");
      setConversations(result);

      if (active) {
        const updated = result.find(item => item.id === active.id);
        if (updated) setActive(updated);
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : "Не удалось загрузить диалоги");
    }
  }, [active]);

  const loadMessages = useCallback(async (conversation: Conversation) => {
    try {
      const after = messagesRef.current[conversation.id] || 0;
      const result = await dmRequest<DirectMessageItem[]>(
        `/${conversation.id}/updates?after_id=${after}`
      );

      if (result.length) {
        setMessages(current => {
          const merged = [...current, ...result];
          const unique = Array.from(
            new Map(merged.map(item => [item.id, item])).values()
          );
          return unique.sort((a, b) => a.id - b.id);
        });

        messagesRef.current[conversation.id] = Math.max(
          after,
          ...result.map(item => item.id)
        );

        const last = Math.max(...result.map(item => item.id));

        void dmRequest(
          `/${conversation.id}/read?message_id=${last}`,
          { method: "POST" }
        );
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : "Ошибка загрузки сообщений");
    }
  }, []);

  const openConversation = async (conversation: Conversation) => {
    setActive(conversation);
    setMessages([]);
    setEditingId(null);
    setDraft("");
    messagesRef.current[conversation.id] = 0;
    setError("");
    setLoading(true);

    try {
      const result = await dmRequest<DirectMessageItem[]>(
        `/${conversation.id}/messages`
      );

      setMessages(result);
      messagesRef.current[conversation.id] = result.length
        ? result[result.length - 1].id
        : 0;

      if (result.length) {
        void dmRequest(
          `/${conversation.id}/read?message_id=${result[result.length - 1].id}`,
          { method: "POST" }
        );
      }
    } catch (e) {
      setError(e instanceof Error ? e.message : "Ошибка загрузки переписки");
    } finally {
      setLoading(false);

          body: JSON.stringify({ friend_id: friendId }),
        }
      );

      const list = await dmRequest<Conversation[]>("/conversations");
      setConversations(list);

      const found = list.find(item => item.id === result.id);

      if (found) {
        await openConversation(found);

      setError(e instanceof Error ? e.message : "Не удалось открыть диалог");
    }

                            ) : (

                              )}
                          </article>
                        );
                      })}

                      <div ref={bottomRef} />
                    </div>

                    <form
                      className="dm-composer"
                      onSubmit={event => {
                        event.preventDefault();
                        void sendMessage();
                      }}
                    >
                      <input
                        value={draft}
                        onChange={event => setDraft(event.target.value)}
                        placeholder={
                          canSend
                            ? "Напишите сообщение..."
                            : "Отправка сообщений недоступна"
                        }
                        maxLength={10000}
                        disabled={!canSend}
                      />
                      <button
                        type="submit"
                        disabled={!draft.trim() || !canSend}
                      >
                        Отправить
                      </button>
                    </form>
                  </>
                ) : (
                  <div className="dm-welcome">
                    <span>✉</span>
                    <strong>Ваши личные сообщения</strong>
                    <p>
                      Выбери диалог слева или начни переписку с другом.
                    </p>
                  </div>
                )}
              </section>
            </div>

            {error && <div className="dm-error">{error}</div>}
          </section>
        </div>
      )}
    </>
  );
}

async function request<T = any>(path: string, options: RequestInit = {}): Promise<T> {
  const headers = new Headers(options.headers);
  headers.set("Accept", "application/json");
  const token = localStorage.getItem("comind_access_token");
  if (token) headers.set("Authorization", `Bearer ${token}`);
  if (options.body && !(options.body instanceof FormData)) {
    headers.set("Content-Type", "application/json");
  }
  const response = await fetch(`${API}${path}`, { ...options, headers });
  const data = response.status === 204 ? null : await response.json().catch(() => ({}));
  if (!response.ok) {
    const detail = data?.detail;
    throw new Error(Array.isArray(detail) ? detail.map((x: any) => x.msg || "").join("; ") : detail || `HTTP ${response.status}`);
  }
  return data as T;
}

                <input value={channelName} onChange={(e) => setChannelName(e.target.value)} placeholder="Название канала: general" minLength={2} maxLength={50} required />
                <select value={newChannelType} onChange={(e) => setNewChannelType(e.target.value)}><option value="text">Текстовый</option><option value="voice">Голосовой</option></select>
                <select
                  value={newChannelVisibility}
                  onChange={(e) => setNewChannelVisibility(e.target.value as "public" | "private")}
                  aria-label="Channel access"
                >
                  <option value="public">{"\u041e\u0442\u043a\u0440\u044b\u0442\u044b\u0439"}</option>
                  <option value="private">{"\u041f\u0440\u0438\u0432\u0430\u0442\u043d\u044b\u0439"}</option>
                </select>
                <button disabled={busy}>Создать</button>
              </form>
              {channels.map((item) => <article className="fp-row" key={item.id}>
                <div className="fp-row-main"><strong>{item.channel_type === "voice" ? "◖" : "#"} {item.name}</strong><small>{item.channel_type || "text"} · ID {item.id}</small></div>
                <div className="fp-row-actions"><button onClick={() => void renameChannel(item)}>Переименовать</button><button className="danger" onClick={() => void removeChannel(item)}>Удалить</button></div>
              </article>)}
            </>}
          </section>}

          {tab === "voice" && <section>
            <h3>Голосовые комнаты</h3>
            <p>Сначала выбери голосовой канал. Сейчас доступны подключение к комнате и список присутствующих; передача живого звука требует отдельного WebRTC-клиента.</p>
            <div className="fp-actions"><button disabled={busy || voiceJoined || !channelId} onClick={() => void joinVoice()}>Подключиться</button><button disabled={busy || !voiceJoined} onClick={() => void leaveVoice()}>Отключиться</button><button disabled={busy || !voiceJoined} onClick={() => void toggleMute()}>{voiceMuted ? "Включить микрофон" : "Выключить микрофон"}</button><button onClick={() => void run(loadVoiceMembers)}>Обновить участников</button></div>
            <p>Состояние: <strong>{voiceJoined ? "подключён" : "не подключён"}</strong> · Микрофон: <strong>{voiceMuted ? "выключен" : "включён"}</strong></p>
            {voiceMembers.map((item) => <article className="fp-row" key={item.user_id}><div className="fp-row-main"><strong>{item.display_name || item.username}</strong><small>{item.muted ? "Микрофон выключен" : "В комнате"}</small></div></article>)}
          </section>}

          {tab === "notifications" && <section>
            <div className="fp-section-title"><h3>Уведомления</h3><button onClick={() => void run(async () => { await request("/api/notifications/read-all", { method: "POST" }); await loadNotifications(); flash("Все уведомления прочитаны."); })}>Прочитать все</button></div>
            {!notifications.length && <p>Уведомлений пока нет.</p>}
            {notifications.map((item) => <article className={`fp-notification ${item.is_read ? "" : "unread"}`} key={item.id}><strong>{item.title}</strong><p>{item.body}</p><small>{item.created_at}</small>{!item.is_read && <button onClick={() => void run(async () => { await request(`/api/notifications/${item.id}/read`, { method: "POST" }); await loadNotifications(); })}>Прочитано</button>}</article>)}
          </section>}

          {tab === "search" && <section>
            <h3>Поиск по COMIND</h3>
            <form className="fp-inline-form" onSubmit={searchAll}><input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Пользователи, сообщества, сообщения" minLength={2} /><button disabled={busy}>Найти</button></form>
            {results && <>
              <h4>Сообщества</h4>{(results.communities || []).map((x: Item) => <article className="fp-row" key={x.id}><div className="fp-row-main"><strong>{x.name}</strong><small>{x.description}</small></div></article>)}
              <h4>Пользователи</h4>{(results.users || []).map((x: Item) => <article className="fp-row" key={x.id}><div className="fp-row-main"><strong>{x.display_name || x.username}</strong><small>@{x.username}</small></div></article>)}
              <h4>Сообщения</h4>{(results.messages || []).map((x: Item) => <article className="fp-row" key={x.id}><div className="fp-row-main"><strong>{x.username} · канал {x.channel_id}</strong><small>{x.content}</small></div></article>)}
            </>}
          </section>}

          {tab === "moderation" && <section>
            <h3>Администрирование и жалобы</h3>
            <p>Список жалоб и решения доступны аккаунту платформенного администратора. Роли в сообществах настраиваются во вкладке «Участники».</p>
            {!reports.length && <p>Открытых жалоб нет или у аккаунта нет доступа к модерации.</p>}
            {reports.map((item) => <article className="fp-row" key={item.id}>
              <div className="fp-row-main"><strong>Жалоба #{item.id} · {item.status}</strong><small>{item.reason} · пользователь {item.target_user_id ?? "не указан"} · сообщение {item.message_id ?? "не указано"}</small></div>
              {item.status === "open" && <div className="fp-row-actions"><button onClick={() => void resolveReport(item, "resolved")}>Решить</button><button className="danger" onClick={() => void resolveReport(item, "rejected")}>Отклонить</button></div>}
            </article>)}
          </section>}
        </div>
      </section>
    </div>
  );
}

async function apiRequest<T>(path: string, options: RequestInit = {}): Promise<T> {
  const headers = new Headers(options.headers);
  headers.set("Accept", "application/json");

  if (options.body) headers.set("Content-Type", "application/json");

  const token = localStorage.getItem("comind_access_token");
  if (token) headers.set("Authorization", `Bearer ${token}`);

  let response: Response;
  try {
    response = await fetch(`${API}${path}`, { ...options, headers });
  } catch {
    throw new Error("Backend недоступен. Запусти сервер COMIND на порту 8000.");
  }

  const data = response.status === 204 ? null : await response.json().catch(() => ({}));

  if (!response.ok) {
    const detail = (data as any)?.detail;
    const message = Array.isArray(detail)
      ? detail.map((x: any) => x.msg || "").join("; ")
      : detail;
    throw new Error(message || `Ошибка HTTP ${response.status}`);
  }

  return data as T;
}

function asArray<T>(value: any): T[] {
  if (Array.isArray(value)) return value;
  if (Array.isArray(value?.items)) return value.items;
  if (Array.isArray(value?.results)) return value.results;
  return [];
}

  async function sendFriendRequest(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const name = friendUsername.trim();
    if (!name) return;
    setFriendNotice(null);
    try {
      const result = await apiRequest<{status: string; username?: string}>("/api/friends/requests", {
        method: "POST",
        body: JSON.stringify({ username: name }),
      });
      setFriendUsername("");
      await loadFriends();
      const labels: Record<string, string> = {
        pending: "Запрос дружбы отправлен.",
        accepted: "Вы теперь друзья.",
        already_pending: "Запрос уже отправлен.",
        already_friends: "Этот пользователь уже в друзьях.",
      };
      const message = labels[result.status] || "Запрос обработан.";
      setFriendNotice({ text: message, error: false });
      notify(message);
    } catch (error) {
      const message = error instanceof Error ? error.message : "Не удалось отправить запрос.";
      setFriendNotice({ text: message, error: true });
      notify(message, true);
    }
  }

  async function renameChannel(item: Channel) {
    const entered = await appPrompt({ title: "\u041f\u0435\u0440\u0435\u0438\u043c\u0435\u043d\u043e\u0432\u0430\u0442\u044c \u043a\u0430\u043d\u0430\u043b", message: "\u0412\u0432\u0435\u0434\u0438\u0442\u0435 \u043d\u043e\u0432\u043e\u0435 \u043d\u0430\u0437\u0432\u0430\u043d\u0438\u0435 \u043a\u0430\u043d\u0430\u043b\u0430:", defaultValue: item.name, placeholder: "\u041d\u043e\u0432\u043e\u0435 \u043d\u0430\u0437\u0432\u0430\u043d\u0438\u0435", confirmText: "\u041f\u0435\u0440\u0435\u0438\u043c\u0435\u043d\u043e\u0432\u0430\u0442\u044c" });
    if (entered === null) return;
    const name = entered.trim();
    if (name.length < 2 || name.length > 50 || !/^[\p{L}\p{N}_ -]+$/u.test(name)) {
      notify("Название: 2–50 символов. Разрешены буквы любого языка, цифры, пробелы, _ и -.", true);
      return;
    }
    try {
      const updated = await apiRequest<{id: number; name: string}>(
        `/api/channels/${item.id}`,
        { method: "PATCH", body: JSON.stringify({ name }) }
      );
      const nextName = updated.name || name;
      setChannels((previous) => previous.map((entry) =>
        entry.id === item.id ? { ...entry, name: nextName } : entry
      ));
      setChannel((previous) =>
        previous && previous.id === item.id ? { ...previous, name: nextName } : previous
      );
      notify("Канал переименован.");
    } catch (error) {
      notify(error instanceof Error ? error.message : "Не удалось переименовать канал.", true);
    }
  }

      {showFeaturePanel && (

          }}

      )}

      <DirectMessages userId={userId} />
    </main>
  );
}

import { useEffect, useRef, useState } from "react";
import "./VoiceRoom.css";

const API = (import.meta.env.VITE_API_URL || "http://127.0.0.1:8000").replace(/\/+$/, "");

const RTC_CONFIG: RTCConfiguration = {
  iceServers: [{ urls: "stun:stun.l.google.com:19302" }],
};

type Peer = { peer_id: string; username: string };
type RemoteAudio = { peerId: string; username: string; stream: MediaStream };
type Meter = { analyser: AnalyserNode; source: MediaStreamAudioSourceNode };

type Signal = {
  type?: string;
  peer_id?: string;
  peers?: Peer[];
  from?: string;
  to?: string;
  username?: string;
  sdp?: RTCSessionDescriptionInit;
  candidate?: RTCIceCandidateInit;
};

function errorMessage(error: unknown): string {
  if (error instanceof DOMException) {
    if (error.name === "NotAllowedError" || error.name === "PermissionDeniedError")
      return "Нет доступа к микрофону. Разреши его в настройках системы.";
    if (error.name === "NotFoundError")
      return "Микрофон не найден.";
    if (error.name === "NotReadableError")
      return "Микрофон занят другой программой.";
  }
  return error instanceof Error ? error.message : "Ошибка голосового подключения.";
}

export default function VoiceRoom({
  channelId,
  channelName,
  username,
  isPrivate = false,
  onInvite,
}: {
  channelId: number;
  channelName: string;
  username: string;
  isPrivate?: boolean;
  onInvite?: () => void;
}) {
  const [joined, setJoined] = useState(false);
  const [busy, setBusy] = useState(false);
  const [muted, setMuted] = useState(false);
  const [deafened, setDeafened] = useState(false);
  const [status, setStatus] = useState("Не подключено");
  const [error, setError] = useState("");
  const [participants, setParticipants] = useState<Peer[]>([]);
  const [remoteAudio, setRemoteAudio] = useState<RemoteAudio[]>([]);
  const [speakingIds, setSpeakingIds] = useState<string[]>([]);
  const [volumes, setVolumes] = useState<Record<string, number>>({});

  const socketRef = useRef<WebSocket | null>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const peersRef = useRef(new Map<string, RTCPeerConnection>());
  const namesRef = useRef(new Map<string, string>());
  const pendingIceRef = useRef(new Map<string, RTCIceCandidateInit[]>());
  const offersRef = useRef(new Set<string>());
  const peerIdRef = useRef("");
  const audioContextRef = useRef<AudioContext | null>(null);
  const metersRef = useRef(new Map<string, Meter>());
  const audioElementsRef = useRef(new Map<string, HTMLAudioElement>());

  function ensureAudioContext(): AudioContext {
    if (!audioContextRef.current || audioContextRef.current.state === "closed") {
      audioContextRef.current = new AudioContext();
    }
    return audioContextRef.current;
  }

  function stopMeter(id: string) {
    const meter = metersRef.current.get(id);
    if (meter) {
      try { meter.source.disconnect(); } catch { /* already disconnected */ }
      try { meter.analyser.disconnect(); } catch { /* already disconnected */ }
      metersRef.current.delete(id);
    }
    setSpeakingIds((old) => old.filter((item) => item !== id));
  }

  function registerMeter(id: string, stream: MediaStream) {
    stopMeter(id);

    const context = ensureAudioContext();
    const analyser = context.createAnalyser();
    analyser.fftSize = 512;
    analyser.smoothingTimeConstant = 0.65;

    const source = context.createMediaStreamSource(stream);
    source.connect(analyser);
    metersRef.current.set(id, { analyser, source });
  }

  function stopAllMeters(closeContext = false) {
    for (const meter of metersRef.current.values()) {
      try { meter.source.disconnect(); } catch { /* already disconnected */ }
      try { meter.analyser.disconnect(); } catch { /* already disconnected */ }
    }

    metersRef.current.clear();
    setSpeakingIds([]);

    if (closeContext) {
      const context = audioContextRef.current;
      audioContextRef.current = null;
      if (context && context.state !== "closed") {
        void context.close().catch(() => {});
      }
    }
  }

  // Измерение уровня аудиосигнала используется только для определения речи.
  // Сырые данные микрофона никуда отдельно не отправляются.
  useEffect(() => {
    const timer = window.setInterval(() => {
      const active: string[] = [];

      for (const [id, meter] of metersRef.current.entries()) {
        const data = new Uint8Array(meter.analyser.fftSize);
        meter.analyser.getByteTimeDomainData(data);

        let sum = 0;
        for (let i = 0; i < data.length; i++) {
          const sample = (data[i] - 128) / 128;
          sum += sample * sample;
        }

        const rms = Math.sqrt(sum / data.length);
        if (rms > 0.025) active.push(id);
      }

      setSpeakingIds((old) => {
        if (
          old.length === active.length &&
          old.every((id) => active.includes(id))
        ) return old;
        return active;
      });
    }, 120);

    return () => window.clearInterval(timer);
  }, []);

  function send(payload: Record<string, unknown>) {
    const socket = socketRef.current;
    if (socket?.readyState === WebSocket.OPEN) {
      socket.send(JSON.stringify(payload));
    }
  }

  function addParticipant(id: string, name: string) {
    if (!id || id === peerIdRef.current) return;

    namesRef.current.set(id, name || "Участник");
    setParticipants((old) => {
      if (old.some((p) => p.peer_id === id)) {
        return old.map((p) =>
          p.peer_id === id ? { ...p, username: name || p.username } : p
        );
      }
      return [...old, { peer_id: id, username: name || "Участник" }];
    });
  }

  function removeParticipant(id: string) {
    peersRef.current.get(id)?.close();
    peersRef.current.delete(id);
    namesRef.current.delete(id);
    pendingIceRef.current.delete(id);
    offersRef.current.delete(id);
    stopMeter(id);

    audioElementsRef.current.delete(id);
    setVolumes((old) => {
      const next = { ...old };
      delete next[id];
      return next;
    });
    setParticipants((old) => old.filter((p) => p.peer_id !== id));
    setRemoteAudio((old) => old.filter((p) => p.peerId !== id));
  }

  function setParticipantVolume(id: string, value: number) {
    const volume = Math.max(0, Math.min(100, Math.round(value)));
    setVolumes((old) => ({ ...old, [id]: volume }));

    const element = audioElementsRef.current.get(id);
    if (element) element.volume = volume / 100;
  }

  function makePeer(id: string, name: string): RTCPeerConnection {
    const existing = peersRef.current.get(id);
    if (existing) return existing;

    addParticipant(id, name);

    const pc = new RTCPeerConnection(RTC_CONFIG);
    const local = streamRef.current;
    local?.getTracks().forEach((track) => pc.addTrack(track, local));

    pc.onicecandidate = (event) => {
      if (event.candidate) {
        send({
          type: "ice-candidate",
          to: id,
          candidate: event.candidate.toJSON(),
        });
      }
    };

    pc.ontrack = (event) => {
      const stream = event.streams[0] || new MediaStream([event.track]);
      const peerName = namesRef.current.get(id) || name;

      registerMeter(id, stream);

      setRemoteAudio((old) => {
        if (old.some((item) => item.peerId === id)) {
          return old.map((item) =>
            item.peerId === id ? { ...item, stream, username: peerName } : item
          );
        }
        return [...old, { peerId: id, username: peerName, stream }];
      });
    };

    pc.onconnectionstatechange = () => {
      if (pc.connectionState === "connected") {
        setStatus("Голосовое соединение активно");
      } else if (pc.connectionState === "failed") {
        setStatus("Не удалось соединиться. Возможно, нужен TURN-сервер.");
      }
    };

    peersRef.current.set(id, pc);
    return pc;
  }

  async function flushIce(id: string, pc: RTCPeerConnection) {
    const queue = pendingIceRef.current.get(id) || [];
    pendingIceRef.current.delete(id);

    for (const candidate of queue) {
      await pc.addIceCandidate(new RTCIceCandidate(candidate));
    }
  }

  async function startOffer(id: string, name: string) {
    if (offersRef.current.has(id)) return;
    offersRef.current.add(id);

    try {
      const pc = makePeer(id, name);
      const offer = await pc.createOffer();
      await pc.setLocalDescription(offer);

      if (pc.localDescription) {
        send({ type: "offer", to: id, sdp: pc.localDescription });
      }
    } catch (e) {
      offersRef.current.delete(id);
      setError(errorMessage(e));
    }
  }

  async function handleSignal(msg: Signal) {
    if (msg.type === "hello") {
      peerIdRef.current = String(msg.peer_id || "");
      const existing = Array.isArray(msg.peers) ? msg.peers : [];

      for (const peer of existing) {
        const id = String(peer.peer_id || "");
        const name = peer.username || "Участник";
        addParticipant(id, name);

        if (peerIdRef.current && peerIdRef.current.localeCompare(id) < 0) {
          void startOffer(id, name);
        }
      }

      setStatus(existing.length ? "Подключение участников..." : "Ожидание игроков");
      return;
    }

    if (msg.type === "peer-joined") {
      const id = String(msg.peer_id || "");
      const name = msg.username || "Участник";
      addParticipant(id, name);

      if (peerIdRef.current && peerIdRef.current.localeCompare(id) < 0) {
        void startOffer(id, name);
      }
      return;
    }

    if (msg.type === "peer-left") {
      removeParticipant(String(msg.peer_id || ""));
      return;
    }

    const id = String(msg.from || "");
    if (!id) return;

    const name = msg.username || namesRef.current.get(id) || "Участник";
    addParticipant(id, name);

    if (msg.type === "offer" && msg.sdp) {
      const pc = makePeer(id, name);
      await pc.setRemoteDescription(new RTCSessionDescription(msg.sdp));
      await flushIce(id, pc);

      const answer = await pc.createAnswer();
      await pc.setLocalDescription(answer);

      if (pc.localDescription) {
        send({ type: "answer", to: id, sdp: pc.localDescription });
      }
      return;
    }

    if (msg.type === "answer" && msg.sdp) {
      const pc = peersRef.current.get(id);
      if (!pc) return;
      await pc.setRemoteDescription(new RTCSessionDescription(msg.sdp));
      await flushIce(id, pc);
      return;
    }

    if (msg.type === "ice-candidate" && msg.candidate) {
      const pc = peersRef.current.get(id);

      if (!pc || !pc.remoteDescription) {
        const queue = pendingIceRef.current.get(id) || [];
        queue.push(msg.candidate);
        pendingIceRef.current.set(id, queue);
      } else {
        await pc.addIceCandidate(new RTCIceCandidate(msg.candidate));
      }
    }
  }

  async function joinVoice() {
    if (busy || joined) return;

    setBusy(true);
    setError("");

    try {
      if (!navigator.mediaDevices?.getUserMedia) {
        throw new Error("Микрофон недоступен в этом контексте приложения.");
      }

      const token = localStorage.getItem("comind_access_token");
      if (!token) throw new Error("Сессия закончилась. Войди в COMIND снова.");

      const context = ensureAudioContext();
      await context.resume();

      const stream = await navigator.mediaDevices.getUserMedia({
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          autoGainControl: true,
        },
        video: false,
      });

      streamRef.current = stream;
      registerMeter("self", stream);

      const response = await fetch(`${API}/api/voice/${channelId}/ticket`, {
        method: "POST",
        headers: {
          Authorization: `Bearer ${token}`,
          Accept: "application/json",
        },
      });

      const ticketData = await response.json().catch(() => ({}));
      if (!response.ok || !ticketData.ticket) {
        throw new Error(
          typeof ticketData.detail === "string"
            ? ticketData.detail
            : `Сервер не выдал билет голосового канала (HTTP ${response.status}).`
        );
      }

      const wsUrl = API.replace(/^http/i, "ws") +
        `/api/voice/ws?ticket=${encodeURIComponent(ticketData.ticket)}`;

      const socket = new WebSocket(wsUrl);
      socketRef.current = socket;

      socket.onmessage = (event) => {
        try {
          const message = JSON.parse(String(event.data)) as Signal;
          void handleSignal(message).catch((e) => setError(errorMessage(e)));
        } catch {
          setError("Сервер прислал некорректное сообщение.");
        }
      };

      socket.onclose = () => {
        if (socketRef.current === socket) {
          socketRef.current = null;
          peersRef.current.forEach((pc) => pc.close());
          peersRef.current.clear();
          streamRef.current?.getTracks().forEach((track) => track.stop());
          streamRef.current = null;
          stopAllMeters(true);
          setRemoteAudio([]);
          setParticipants([]);
          setJoined(false);
          setStatus("Сигнализация отключена");
        }
      };

      await new Promise<void>((resolve, reject) => {
        const timeout = window.setTimeout(() => {
          reject(new Error("Тайм-аут подключения к голосовому серверу."));
          socket.close();
        }, 10000);

        socket.onopen = () => {
          window.clearTimeout(timeout);
          resolve();
        };

        socket.onerror = () => {
          window.clearTimeout(timeout);
          reject(new Error("Не удалось открыть WebSocket. Проверь запуск backend."));
        };
      });

      setJoined(true);
      setStatus("Подключено; ожидаем участников");
    } catch (e) {
      socketRef.current?.close();
      socketRef.current = null;
      peersRef.current.forEach((pc) => pc.close());
      peersRef.current.clear();
      streamRef.current?.getTracks().forEach((track) => track.stop());
      streamRef.current = null;
      stopAllMeters(true);
      setJoined(false);
      setStatus("Не подключено");
      setError(errorMessage(e));
    } finally {
      setBusy(false);
    }
  }

  function leaveVoice() {
    const socket = socketRef.current;
    socketRef.current = null;

    if (socket) {
      if (socket.readyState === WebSocket.OPEN) {
        try { socket.send(JSON.stringify({ type: "leave" })); } catch { /* closing */ }
      }
      socket.close();
    }

    peersRef.current.forEach((pc) => pc.close());
    peersRef.current.clear();

    streamRef.current?.getTracks().forEach((track) => track.stop());
    streamRef.current = null;

    stopAllMeters(true);
    namesRef.current.clear();
    pendingIceRef.current.clear();
    offersRef.current.clear();
    peerIdRef.current = "";

    setRemoteAudio([]);
    setParticipants([]);
    setJoined(false);
    setMuted(false);
    setDeafened(false);
    setStatus("Не подключено");
  }

  function toggleMute() {
    const next = !muted;
    streamRef.current?.getAudioTracks().forEach((track) => {
      track.enabled = !next;
    });
    setMuted(next);
  }

  useEffect(() => {
    return () => {
      socketRef.current?.close();
      socketRef.current = null;

      peersRef.current.forEach((pc) => pc.close());
      peersRef.current.clear();

      streamRef.current?.getTracks().forEach((track) => track.stop());
      streamRef.current = null;
      stopAllMeters(true);
    };
  }, []);

  return (
    <div className="voice-room">
      <header className="voice-room-header">
        <div>
          <span className="voice-eyebrow">COMIND / WEBRTC</span>
          <h3>Голосовой канал: {channelName}</h3>
          <p>Участников: {participants.length + 1}</p>
        </div>
        <span className={`voice-status ${joined ? "is-live" : ""}`}>
          <i />{status}
        </span>
      </header>

      {error && <div className="voice-error" role="alert">{error}</div>}

      <div className="voice-participants">
        <div className={`voice-person self-person ${speakingIds.includes("self") ? "is-speaking" : ""}`}>
          <div className="voice-avatar">{username.slice(0, 1).toUpperCase()}</div>

          <div className="voice-person-copy">
            <strong>{username} (вы)</strong>
            <span>{joined ? (muted ? "Микрофон выключен" : "Микрофон включён") : "Не подключено"}</span>
          </div>

          <div className="voice-badges">
            {joined && (
              <span className={`voice-badge ${speakingIds.includes("self") && !muted ? "speaking" : muted ? "disabled" : "ready"}`}>
                {muted ? "МИКРОФОН ВЫКЛ" : speakingIds.includes("self") ? "ВЫ ГОВОРИТЕ" : "МИКРОФОН ВКЛ"}
              </span>
            )}
            {joined && deafened && (
              <span className="voice-badge disabled">ЗВУК ВЫКЛ</span>
            )}
          </div>
        </div>

        {participants.map((peer) => {
          const speaking = speakingIds.includes(peer.peer_id);
          const volume = volumes[peer.peer_id] ?? 100;

          return (
            <div
              className={`voice-person remote-person ${speaking ? "is-speaking" : ""}`}
              key={peer.peer_id}
            >
              <div className="voice-avatar remote-avatar">
                {peer.username.slice(0, 1).toUpperCase()}
              </div>

              <div className="voice-person-copy">
                <strong>{peer.username}</strong>
                <span>{speaking ? "Сейчас говорит" : "Голосовой участник"}</span>
              </div>

              <span className={`voice-badge ${speaking ? "speaking" : "ready"}`}>
                {speaking ? "ГОВОРИТ" : "В КОМНАТЕ"}
              </span>

              <div className="voice-volume">
                <label htmlFor={`volume-${peer.peer_id}`}>
                  <span>Громкость участника</span>
                  <strong>{volume}%</strong>
                </label>

                <input
                  id={`volume-${peer.peer_id}`}
                  aria-label={`Громкость участника ${peer.username}`}
                  type="range"
                  min="0"
                  max="100"
                  step="1"
                  value={volume}
                  onChange={(event) => setParticipantVolume(peer.peer_id, Number(event.target.value))}
                />
              </div>
            </div>
          );
        })}

        {joined && participants.length === 0 && (
          <div className="voice-empty">Ожидание других участников комнаты.</div>
        )}
      </div>

      <div className="voice-controls">
        {isPrivate && onInvite && (
          <button className="voice-button" type="button" onClick={onInvite}>
            {"\u041f\u0440\u0438\u0433\u043b\u0430\u0441\u0438\u0442\u044c \u0443\u0447\u0430\u0441\u0442\u043d\u0438\u043a\u0430"}
          </button>
        )}
        {!joined ? (
          <button className="voice-join" onClick={() => void joinVoice()} disabled={busy}>
            {busy ? "Подключение..." : "Подключиться к голосу"}
          </button>
        ) : (
          <>
            <button className={`voice-button ${muted ? "selected" : ""}`} onClick={toggleMute}>
              {muted ? "Включить микрофон" : "Выключить микрофон"}
            </button>

            <button
              className={`voice-button ${deafened ? "selected" : ""}`}
              onClick={() => setDeafened((value) => !value)}
            >
              {deafened ? "Включить звук" : "Выключить звук"}
            </button>

            <button className="voice-leave" onClick={leaveVoice}>Отключиться</button>
          </>
        )}
      </div>

      <div className="voice-audio-hidden">
        {remoteAudio.map((item) => (
          <audio
            key={item.peerId}
            autoPlay
            playsInline
            muted={deafened}
            ref={(element) => {
              if (!element) {
                audioElementsRef.current.delete(item.peerId);
                return;
              }

              audioElementsRef.current.set(item.peerId, element);
              element.volume = (volumes[item.peerId] ?? 100) / 100;

              if (element.srcObject !== item.stream) {
                element.srcObject = item.stream;
                void element.play().catch(() => {
                  setError("Не удалось воспроизвести звук. Проверь настройки вывода аудио.");
                });
              }
            }}
          />
        ))}
      </div>
    </div>
  );
}
